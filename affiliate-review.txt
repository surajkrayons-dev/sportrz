import { NextRequest, NextResponse } from 'next/server';
import { getPool } from '@/lib/mysql';
import type { RowDataPacket } from 'mysql2/promise';
import { getCurrentUser } from '@/lib/session';
import { getRazorpayConfig } from '@/lib/payment-settings';

export const runtime = 'nodejs';

export async function POST(request: NextRequest) {
  try {
    const user = await getCurrentUser();
    if (!user) return NextResponse.json({ error: 'Sign in before purchasing a plan.' }, { status: 401 });

    const body = await request.json();
    const planCode = String(body.planId || '').trim().toLowerCase();
    if (!/^[a-z0-9_-]{1,50}$/.test(planCode)) {
      return NextResponse.json({ error: 'Please select a valid subscription plan.' }, { status: 400 });
    }

    const [planRows] = await getPool().execute<any[]>(
      `SELECT id, code, name, price_paise, currency, duration_days
       FROM plans WHERE code = ? AND is_active = 1 LIMIT 1`,
      [planCode],
    );
    const plan = planRows[0];
    if (!plan) {
      return NextResponse.json({ error: 'This plan is currently disabled. Please contact the site administrator.' }, { status: 409 });
    }
    const amountPaise = Number(plan.price_paise);
    const durationDays = plan.duration_days == null ? null : Number(plan.duration_days);
    if (!Number.isSafeInteger(amountPaise) || amountPaise <= 0 || (durationDays !== null && (!Number.isInteger(durationDays) || durationDays < 1 || durationDays > 3650))) {
      return NextResponse.json({ error: 'The selected plan has an invalid price or duration. Please contact the site administrator.' }, { status: 409 });
    }

    const razorpayConfig = await getRazorpayConfig();
    const razorpayKeyId = razorpayConfig?.keyId;
    const razorpaySecret = razorpayConfig?.secret;
    if (!razorpayKeyId || !razorpaySecret) {
      return NextResponse.json({ error: 'Razorpay credentials are not configured on the server.' }, { status: 500 });
    }

    const currency = String(plan.currency || 'INR').toUpperCase();

    // Keep an auditable attribution snapshot on the Razorpay Order itself.
    // This data is generated server-side from MySQL; it is not accepted from the browser.
    const [referralRows] = await getPool().execute<Array<RowDataPacket & {
      affiliate_user_id: string;
      referral_code: string;
      affiliate_email: string;
      affiliate_name: string | null;
    }>>(
      `SELECT ar.affiliate_user_id, ar.referral_code,
              au.email AS affiliate_email, au.display_name AS affiliate_name
         FROM affiliate_referrals ar
         JOIN affiliate_profiles ap ON ap.user_id = ar.affiliate_user_id
         JOIN users au ON au.id = ar.affiliate_user_id
        WHERE ar.referred_user_id = ? AND ar.status = 'qualified'
        LIMIT 1`,
      [user.uid],
    );
    const affiliate = referralRows[0];
    const paymentNotes: Record<string, string> = {
      app: 'Sportrz',
      app_user_id: String(user.uid).slice(0, 256),
      customer_email: String(user.email || '').slice(0, 256),
      customer_name: String(user.displayName || '').slice(0, 256),
      plan_code: String(plan.code).slice(0, 256),
      plan_name: String(plan.name).slice(0, 256),
      affiliate_user_id: String(affiliate?.affiliate_user_id || 'none').slice(0, 256),
      affiliate_code: String(affiliate?.referral_code || 'none').slice(0, 256),
      affiliate_email: String(affiliate?.affiliate_email || 'none').slice(0, 256),
      affiliate_name: String(affiliate?.affiliate_name || 'none').slice(0, 256),
    };

    // Razorpay supports up to 15 note key/value pairs. Keep the full attribution
    // on the server-created Order and send only the useful checkout notes to the browser.
    const checkoutNotes: Record<string, string> = {
      app_user_id: paymentNotes.app_user_id,
      customer_email: paymentNotes.customer_email,
      plan_code: paymentNotes.plan_code,
      affiliate_code: paymentNotes.affiliate_code,
    };
    const orderPayload = {
      amount: amountPaise,
      currency,
      receipt: `sp_${Date.now()}`.slice(0, 40),
      notes: paymentNotes,
    };
    const authorization = Buffer.from(`${razorpayKeyId}:${razorpaySecret}`).toString('base64');
    const response = await fetch('https://api.razorpay.com/v1/orders', {
      method: 'POST',
      headers: { Authorization: `Basic ${authorization}`, 'Content-Type': 'application/json' },
      body: JSON.stringify(orderPayload),
    });

    const order = await response.json();
    if (!response.ok) {
      console.error('Razorpay order creation failed:', order);
      return NextResponse.json({ error: 'Razorpay could not create the order.' }, { status: response.status });
    }

    await getPool().execute(
      `INSERT INTO payments (user_id, plan_id, provider, order_id, amount_paise, currency, status)
       VALUES (?, ?, 'razorpay', ?, ?, ?, 'created')`,
      [user.uid, Number(plan.id), order.id, amountPaise, currency],
    );

    return NextResponse.json({
      success: true,
      order: { id: order.id, amount: order.amount, currency: order.currency, receipt: order.receipt },
      keyId: razorpayKeyId,
      plan: { id: plan.code, name: plan.name, durationDays },
      checkoutNotes,
    });
  } catch (error) {
    console.error('Create order failed:', error);
    return NextResponse.json({ error: 'Unable to create the payment order.' }, { status: 500 });
  }
}

import { NextRequest, NextResponse } from 'next/server';
import { createHmac, timingSafeEqual } from 'node:crypto';
import { getPool } from '@/lib/mysql';
import { getCurrentUser } from '@/lib/session';
import { getRazorpayConfig } from '@/lib/payment-settings';
import { ensureAffiliateCommissionForPayment } from '@/lib/affiliate-commission';

export const runtime = 'nodejs';

export async function POST(request: NextRequest) {
  try {
    const user = await getCurrentUser();
    if (!user) return NextResponse.json({ error: 'Sign in before verifying a payment.' }, { status: 401 });

    const body = await request.json();
    const orderId = String(body.razorpay_order_id || '');
    const paymentId = String(body.razorpay_payment_id || '');
    const signature = String(body.razorpay_signature || '');
    if (!orderId || !paymentId || !signature) {
      return NextResponse.json({ error: 'Missing required payment details.' }, { status: 400 });
    }

    const razorpayConfig = await getRazorpayConfig();
    const razorpayKeyId = razorpayConfig?.keyId;
    const razorpaySecret = razorpayConfig?.secret;
    if (!razorpayKeyId || !razorpaySecret) {
      return NextResponse.json({ error: 'Razorpay credentials are not configured on the server.' }, { status: 500 });
    }

    const [orderRows] = await getPool().execute<any[]>(
      `SELECT p.id, p.user_id, p.plan_id, p.amount_paise, p.currency, p.status,
              p.order_id, p.provider_payment_id, pl.code AS plan_code, pl.duration_days
       FROM payments p
       INNER JOIN plans pl ON pl.id = p.plan_id
       WHERE p.order_id = ? AND p.user_id = ? AND p.provider = 'razorpay'
       LIMIT 1`,
      [orderId, user.uid],
    );
    const order = orderRows[0];
    if (!order) return NextResponse.json({ error: 'Payment order not found for this account.' }, { status: 404 });

    // A previous deployment or a transient issue may have captured the payment
    // before the commission was written. Reconcile idempotently on every retry.
    if (order.status === 'captured') {
      const connection = await getPool().getConnection();
      try {
        await connection.beginTransaction();
        await ensureAffiliateCommissionForPayment(connection, Number(order.id));
        await connection.commit();
      } catch (error) {
        await connection.rollback();
        throw error;
      } finally {
        connection.release();
      }
      return NextResponse.json({ success: true, message: 'Payment already verified.' });
    }
    if (order.status !== 'created' && order.status !== 'authorized') {
      return NextResponse.json({ error: 'This payment order is not payable.' }, { status: 409 });
    }

    const expectedSignature = createHmac('sha256', razorpaySecret).update(`${orderId}|${paymentId}`).digest('hex');
    const expectedBuffer = Buffer.from(expectedSignature, 'hex');
    const actualBuffer = Buffer.from(signature, 'hex');
    if (expectedBuffer.length !== actualBuffer.length || !timingSafeEqual(expectedBuffer, actualBuffer)) {
      return NextResponse.json({ error: 'Payment signature verification failed.' }, { status: 400 });
    }

    const authorization = Buffer.from(`${razorpayKeyId}:${razorpaySecret}`).toString('base64');
    const paymentResponse = await fetch(`https://api.razorpay.com/v1/payments/${encodeURIComponent(paymentId)}`, {
      headers: { Authorization: `Basic ${authorization}` },
    });
    const paymentData = await paymentResponse.json();
    if (!paymentResponse.ok) {
      console.error('Razorpay payment lookup failed:', paymentData);
      return NextResponse.json({ error: 'Unable to verify payment with Razorpay.' }, { status: 502 });
    }
    if (paymentData.order_id !== orderId || paymentData.status !== 'captured' || Number(paymentData.amount) !== Number(order.amount_paise) || paymentData.currency !== order.currency) {
      return NextResponse.json({ error: 'Payment details do not match the recorded order.' }, { status: 400 });
    }

    const durationDays = Number(order.duration_days);
    if (!Number.isInteger(durationDays) || durationDays < 1 || durationDays > 3650) {
      return NextResponse.json({ error: 'The purchased plan duration is invalid.' }, { status: 500 });
    }

    const pool = getPool();
    const connection = await pool.getConnection();
    try {
      await connection.beginTransaction();
      const [lockedRows] = await connection.execute<any[]>(
        'SELECT status FROM payments WHERE id = ? AND user_id = ? FOR UPDATE',
        [order.id, user.uid],
      );
      if (!lockedRows.length) throw new Error('Payment order disappeared during verification.');
      if (lockedRows[0].status === 'captured') {
        await ensureAffiliateCommissionForPayment(connection, Number(order.id));
        await connection.commit();
        return NextResponse.json({ success: true, message: 'Payment already verified.' });
      }

      await connection.execute(
        `UPDATE payments
         SET status = 'captured', provider_payment_id = ?, paid_at = UTC_TIMESTAMP(), provider_payload = ?, updated_at = UTC_TIMESTAMP()
         WHERE id = ?`,
        [paymentId, JSON.stringify(paymentData), order.id],
      );
      await connection.execute(
        `UPDATE subscriptions SET status = 'expired', updated_at = UTC_TIMESTAMP()
         WHERE user_id = ? AND status = 'active' AND end_at IS NOT NULL AND end_at <= UTC_TIMESTAMP()`,
        [user.uid],
      );
      await connection.execute(
        `UPDATE subscriptions SET status = 'cancelled', updated_at = UTC_TIMESTAMP()
         WHERE user_id = ? AND status = 'active'`,
        [user.uid],
      );
      await connection.execute(
        `INSERT INTO subscriptions (user_id, plan_id, payment_record_id, status, source, start_at, end_at)
         VALUES (?, ?, ?, 'active', 'purchase', UTC_TIMESTAMP(), DATE_ADD(UTC_TIMESTAMP(), INTERVAL ${durationDays} DAY))`,
        [user.uid, Number(order.plan_id), Number(order.id)],
      );
      // Create a commission snapshot using the same idempotent path used by retries.
      await ensureAffiliateCommissionForPayment(connection, Number(order.id));
      await connection.commit();
    } catch (error) {
      await connection.rollback();
      throw error;
    } finally {
      connection.release();
    }

    const [subscriptionRows] = await pool.execute<any[]>(
      `SELECT start_at, end_at FROM subscriptions
       WHERE user_id = ? AND payment_record_id = ? ORDER BY id DESC LIMIT 1`,
      [user.uid, Number(order.id)],
    );
    const savedSubscription = subscriptionRows[0];
    return NextResponse.json({
      success: true,
      message: 'Payment verified and subscription activated.',
      subscription: {
        plan: order.plan_code,
        paymentId,
        orderId,
        startDate: savedSubscription?.start_at ? new Date(savedSubscription.start_at).toISOString() : new Date().toISOString(),
        endDate: savedSubscription?.end_at ? new Date(savedSubscription.end_at).toISOString() : new Date(Date.now() + durationDays * 86400000).toISOString(),
        isActive: true,
      },
    });
  } catch (error) {
    console.error('Payment verification failed:', error);
    return NextResponse.json({ error: 'Payment verification failed. Please contact support if money was deducted.' }, { status: 500 });
  }
}

import { createHmac, timingSafeEqual } from 'node:crypto';
import { NextRequest, NextResponse } from 'next/server';
import { getRazorpayConfig } from '@/lib/payment-settings';
import { finalizeCapturedRazorpayPayment, PaymentSettlementError } from '@/lib/razorpay-settlement';

export const runtime = 'nodejs';

export async function POST(request: NextRequest) {
  try {
    const config = await getRazorpayConfig();
    if (!config?.keyId || !config.secret || !config.webhookSecret) {
      return NextResponse.json({ error: 'Razorpay webhook secret is not configured.' }, { status: 503 });
    }

    const signature = request.headers.get('x-razorpay-signature') || '';
    if (!/^[a-f0-9]{64}$/i.test(signature)) return NextResponse.json({ error: 'Invalid webhook signature.' }, { status: 400 });
    const rawBody = await request.text();
    if (!rawBody || rawBody.length > 1_000_000) return NextResponse.json({ error: 'Invalid webhook payload.' }, { status: 400 });
    const expected = createHmac('sha256', config.webhookSecret).update(rawBody, 'utf8').digest();
    const received = Buffer.from(signature, 'hex');
    if (expected.length !== received.length || !timingSafeEqual(expected, received)) {
      return NextResponse.json({ error: 'Webhook signature verification failed.' }, { status: 400 });
    }

    let event: any;
    try { event = JSON.parse(rawBody); }
    catch { return NextResponse.json({ error: 'Webhook body must be valid JSON.' }, { status: 400 }); }
    // Only process the final captured event. Other configured events are acknowledged and ignored.
    if (event.event !== 'payment.captured') return NextResponse.json({ received: true, ignored: true });
    const webhookPayment = event?.payload?.payment?.entity;
    if (!webhookPayment?.id || !webhookPayment?.order_id || webhookPayment.status !== 'captured') {
      return NextResponse.json({ error: 'The captured payment entity is missing required fields.' }, { status: 400 });
    }

    // Re-fetch the payment from Razorpay after authenticating the webhook, then reconcile the exact order, currency and amount.
    const authorization = Buffer.from(`${config.keyId}:${config.secret}`).toString('base64');
    const providerResponse = await fetch(`https://api.razorpay.com/v1/payments/${encodeURIComponent(String(webhookPayment.id))}`, {
      headers: { Authorization: `Basic ${authorization}` }, cache: 'no-store',
    });
    const payment = await providerResponse.json();
    if (!providerResponse.ok) {
      console.error('Razorpay webhook payment lookup failed:', { status: providerResponse.status, paymentId: webhookPayment.id, response: payment });
      return NextResponse.json({ error: 'Could not reconcile this payment with Razorpay. The provider may retry the webhook.' }, { status: 502 });
    }
    if (payment.id !== webhookPayment.id || payment.order_id !== webhookPayment.order_id || payment.status !== 'captured') {
      return NextResponse.json({ error: 'The provider payment does not match the webhook event.' }, { status: 409 });
    }

    const result = await finalizeCapturedRazorpayPayment({
      orderId: String(payment.order_id), paymentId: String(payment.id), amount: Number(payment.amount), currency: String(payment.currency),
      status: String(payment.status), rawPayload: payment,
    });
    return NextResponse.json({ received: true, reconciled: true, alreadyApplied: result.alreadyApplied });
  } catch (error) {
    if (error instanceof PaymentSettlementError) return NextResponse.json({ error: error.message }, { status: error.status });
    console.error('Razorpay webhook processing failed:', error);
    return NextResponse.json({ error: 'Webhook processing failed. The provider may retry the event.' }, { status: 500 });
  }
}

import mysql, { type Pool, type PoolOptions } from 'mysql2/promise';

type DatabaseGlobals = typeof globalThis & { __sportrzMysqlPool?: Pool };
const globals = globalThis as DatabaseGlobals;

function getDatabaseOptions(): PoolOptions {
  const databaseUrl = process.env.DATABASE_URL;

  if (databaseUrl) {
    const parsed = new URL(databaseUrl);
    const options: PoolOptions = {
      host: parsed.hostname,
      port: parsed.port ? Number(parsed.port) : 3306,
      user: decodeURIComponent(parsed.username),
      password: decodeURIComponent(parsed.password),
      database: parsed.pathname.replace(/^\//, ''),
      waitForConnections: true,
      connectionLimit: 10,
      queueLimit: 0,
      decimalNumbers: true,
      timezone: 'Z',
      enableKeepAlive: true,
    };

    if (parsed.searchParams.get('ssl') === 'true') {
      options.ssl = { rejectUnauthorized: true };
    }

    return options;
  }

  const host = process.env.MYSQL_HOST;
  const user = process.env.MYSQL_USER;
  const database = process.env.MYSQL_DATABASE;
  if (!host || !user || !database) {
    throw new Error(
      'MySQL is not configured. Set DATABASE_URL or MYSQL_HOST, MYSQL_USER, MYSQL_PASSWORD, MYSQL_DATABASE, and MYSQL_PORT in .env.local.'
    );
  }

  return {
    host,
    port: Number(process.env.MYSQL_PORT || 3306),
    user,
    password: process.env.MYSQL_PASSWORD || '',
    database,
    waitForConnections: true,
    connectionLimit: 10,
    queueLimit: 0,
    decimalNumbers: true,
    timezone: 'Z',
    enableKeepAlive: true,
    ...(process.env.MYSQL_SSL === 'true' ? { ssl: { rejectUnauthorized: true } } : {}),
  };
}

export function getPool(): Pool {
  if (!globals.__sportrzMysqlPool) {
    globals.__sportrzMysqlPool = mysql.createPool(getDatabaseOptions());
  }
  return globals.__sportrzMysqlPool;
}

export async function checkDatabaseConnection(): Promise<void> {
  const connection = await getPool().getConnection();
  try {
    await connection.ping();
  } finally {
    connection.release();
  }
}

export function parseJson<T>(value: unknown, fallback: T): T {
  if (value == null) return fallback;
  if (typeof value !== 'string') return value as T;
  try {
    return JSON.parse(value) as T;
  } catch {
    return fallback;
  }
}

declare global {
  interface Window {
    Razorpay: any;
  }
}

interface RazorpayOptions {
  key: string;
  amount: number;
  currency: string;
  name: string;
  description: string;
  order_id: string;
  handler: (response: RazorpayResponse) => void;
  prefill?: {
    name?: string;
    email?: string;
    contact?: string;
  };
  notes?: Record<string, string>;
  theme?: {
    color?: string;
    backdrop_color?: string;
  };
  modal?: {
    backdropclose?: boolean;
    escape?: boolean;
    handleback?: boolean;
    ondismiss?: () => void;
  };
}

interface RazorpayResponse {
  razorpay_payment_id: string;
  razorpay_order_id: string;
  razorpay_signature: string;
}

export const loadRazorpayScript = (): Promise<boolean> => {
  return new Promise((resolve) => {
    // Check if script is already loaded
    if (window.Razorpay) {
      resolve(true);
      return;
    }

    const script = document.createElement('script');
    script.src = 'https://checkout.razorpay.com/v1/checkout.js';
    script.onload = () => resolve(true);
    script.onerror = () => resolve(false);
    document.body.appendChild(script);
  });
};

export const initiateRazorpayPayment = async (
  options: RazorpayOptions
): Promise<void> => {
  const isScriptLoaded = await loadRazorpayScript();
  
  if (!isScriptLoaded) {
    throw new Error('Failed to load Razorpay script');
  }

  const razorpay = new window.Razorpay(options);
  razorpay.open();
};

export const createRazorpayOrder = async (planId: string) => {
  const response = await fetch('/api/create-order', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ planId }),
  });

  const data = await response.json();
  if (!response.ok) throw new Error(data.error || 'Failed to create order');
  return data;
};

export const verifyRazorpayPayment = async (
  razorpay_order_id: string,
  razorpay_payment_id: string,
  razorpay_signature: string,
) => {
  const response = await fetch('/api/verify-payment', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ razorpay_order_id, razorpay_payment_id, razorpay_signature }),
  });

  const data = await response.json();
  if (!response.ok) throw new Error(data.error || 'Payment verification failed');
  return data;
};

import type { RowDataPacket } from 'mysql2/promise';
import { getPool } from '@/lib/mysql';

export class PaymentSettlementError extends Error {
  status: number;
  constructor(message: string, status = 400) { super(message); this.name = 'PaymentSettlementError'; this.status = status; }
}

export interface CapturedRazorpayPayment {
  orderId: string;
  paymentId: string;
  amount: number;
  currency: string;
  status: string;
  rawPayload: unknown;
  expectedUserId?: string;
}

export async function finalizeCapturedRazorpayPayment(input: CapturedRazorpayPayment) {
  const pool = getPool();
  const connection = await pool.getConnection();
  try {
    await connection.beginTransaction();
    const query = `SELECT p.id, p.user_id, p.plan_id, p.amount_paise, p.currency, p.status, p.provider_payment_id,
                          pl.code AS plan_code, pl.duration_days
                   FROM payments p JOIN plans pl ON pl.id = p.plan_id
                   WHERE p.order_id = ? AND p.provider = 'razorpay' ${input.expectedUserId ? 'AND p.user_id = ?' : ''}
                   LIMIT 1 FOR UPDATE`;
    const args = input.expectedUserId ? [input.orderId, input.expectedUserId] : [input.orderId];
    const [orders] = await connection.execute<Array<RowDataPacket & {
      id: number; user_id: string; plan_id: number; amount_paise: number; currency: string; status: string;
      provider_payment_id: string | null; plan_code: string; duration_days: number | null;
    }>>(query, args);
    const order = orders[0];
    if (!order) throw new PaymentSettlementError('Payment order not found.', 404);
    if (input.status !== 'captured' || input.orderId.length === 0 || input.paymentId.length === 0) throw new PaymentSettlementError('Payment is not captured.');
    if (input.amount !== Number(order.amount_paise) || input.currency !== order.currency) throw new PaymentSettlementError('Payment amount or currency does not match the order.');
    if (order.provider_payment_id && order.provider_payment_id !== input.paymentId) throw new PaymentSettlementError('A different payment is already attached to this order.', 409);
    if (!['created', 'authorized', 'captured'].includes(order.status)) throw new PaymentSettlementError('This order is not payable.', 409);

    const durationDays = order.duration_days == null ? null : Number(order.duration_days);
    if (durationDays !== null && (!Number.isInteger(durationDays) || durationDays < 1 || durationDays > 3650)) throw new PaymentSettlementError('The purchased plan duration is invalid.', 500);

    if (order.status !== 'captured' || !order.provider_payment_id) {
      await connection.execute(
        `UPDATE payments SET status = 'captured', provider_payment_id = ?, paid_at = UTC_TIMESTAMP(), provider_payload = ?, updated_at = UTC_TIMESTAMP() WHERE id = ?`,
        [input.paymentId, JSON.stringify(input.rawPayload), order.id],
      );
    }

    const [existingSubscriptionRows] = await connection.execute<Array<RowDataPacket & { id: number; start_at: Date | string | null; end_at: Date | string | null }>>(
      'SELECT id, start_at, end_at FROM subscriptions WHERE payment_record_id = ? ORDER BY id DESC LIMIT 1 FOR UPDATE', [order.id],
    );
    let savedSubscription = existingSubscriptionRows[0];
    if (!savedSubscription) {
      await connection.execute(
        `UPDATE subscriptions SET status = 'expired', updated_at = UTC_TIMESTAMP()
         WHERE user_id = ? AND status = 'active' AND end_at IS NOT NULL AND end_at <= UTC_TIMESTAMP()`, [order.user_id],
      );
      await connection.execute(
        `UPDATE subscriptions SET status = 'cancelled', updated_at = UTC_TIMESTAMP() WHERE user_id = ? AND status = 'active'`, [order.user_id],
      );
      await connection.execute(
        `INSERT INTO subscriptions (user_id, plan_id, payment_record_id, status, source, start_at, end_at)
         VALUES (?, ?, ?, 'active', 'purchase', UTC_TIMESTAMP(), ${durationDays === null ? 'NULL' : 'DATE_ADD(UTC_TIMESTAMP(), INTERVAL ? DAY)'})`,
        durationDays === null ? [order.user_id, Number(order.plan_id), Number(order.id)] : [order.user_id, Number(order.plan_id), Number(order.id), durationDays],
      );
      const [newRows] = await connection.execute<Array<RowDataPacket & { id: number; start_at: Date | string | null; end_at: Date | string | null }>>(
        'SELECT id, start_at, end_at FROM subscriptions WHERE payment_record_id = ? ORDER BY id DESC LIMIT 1', [order.id],
      );
      savedSubscription = newRows[0];
    }

    // This is idempotent per payment_record_id. The stored percentage is a snapshot of the profile rate at capture time.
    await connection.execute(
      `INSERT IGNORE INTO affiliate_commissions (affiliate_user_id, referred_user_id, payment_record_id, amount_paise, commission_percent, commission_paise, status)
       SELECT ar.affiliate_user_id, ar.referred_user_id, p.id, p.amount_paise, ap.commission_percent,
              FLOOR(p.amount_paise * ap.commission_percent / 100), 'pending'
       FROM payments p
       JOIN affiliate_referrals ar ON ar.referred_user_id = p.user_id AND ar.status = 'qualified' AND COALESCE(p.paid_at, p.created_at) >= ar.created_at
       JOIN affiliate_profiles ap ON ap.user_id = ar.affiliate_user_id AND ap.status = 'active'
       JOIN users au ON au.id = ap.user_id AND au.account_role = 'affiliate' AND au.status = 'active'
       WHERE p.id = ? LIMIT 1`, [Number(order.id)],
    );

    await connection.commit();
    const toIso = (value: Date | string | null | undefined): string | null => {
      if (!value) return null;
      return value instanceof Date ? value.toISOString() : new Date(value).toISOString();
    };
    return {
      alreadyApplied: Boolean(existingSubscriptionRows[0]),
      userId: order.user_id,
      planCode: order.plan_code,
      paymentId: input.paymentId,
      orderId: input.orderId,
      subscription: {
        startDate: toIso(savedSubscription?.start_at) || new Date().toISOString(),
        endDate: toIso(savedSubscription?.end_at),
        isActive: true,
      },
    };
  } catch (error) {
    try { await connection.rollback(); } catch { /* Ignore rollback after a failed transaction/commit. */ }
    throw error;
  } finally { connection.release(); }
}

function getSessionSecret(): string {
  const secret = process.env.AUTH_SECRET;
  if (secret) return secret;
  if (process.env.NODE_ENV === 'production') {
    throw new Error('AUTH_SECRET must be configured in production.');
  }
  return 'local-development-secret-change-before-deployment';
}

function signSessionPayload(payload: string): string {
  return createHmac('sha256', getSessionSecret()).update(payload).digest('base64url');
}

export function createSessionToken(userId: string): string {
  const expiresAt = Math.floor(Date.now() / 1000) + SESSION_LIFETIME_SECONDS;
  const payload = `${userId}.${expiresAt}`;
  return `${payload}.${signSessionPayload(payload)}`;
}

function readSessionToken(token: string | undefined): string | null {
  if (!token) return null;
  const parts = token.split('.');
  if (parts.length !== 3) return null;

  const [userId, expiresRaw, signature] = parts;
  const expiresAt = Number(expiresRaw);
  if (!userId || !Number.isFinite(expiresAt) || expiresAt <= Math.floor(Date.now() / 1000)) return null;

  const payload = `${userId}.${expiresRaw}`;
  const expected = Buffer.from(signSessionPayload(payload));
  const actual = Buffer.from(signature);
  if (expected.length !== actual.length || !timingSafeEqual(expected, actual)) return null;
  return userId;
}

export function setSessionCookie(response: NextResponse, userId: string): void {
  response.cookies.set(SESSION_COOKIE, createSessionToken(userId), {
    httpOnly: true,
    secure: process.env.NODE_ENV === 'production',
    sameSite: 'lax',
    path: '/',
    maxAge: SESSION_LIFETIME_SECONDS,
  });
}

export function clearSessionCookie(response: NextResponse): void {
  response.cookies.set(SESSION_COOKIE, '', {
    httpOnly: true,
    secure: process.env.NODE_ENV === 'production',
    sameSite: 'lax',
    path: '/',
    maxAge: 0,
  });
}

export async function getCurrentUser(): Promise<AppUser | null> {
  const cookieStore = await cookies();
  const userId = readSessionToken(cookieStore.get(SESSION_COOKIE)?.value);
  if (!userId) return null;

  const [rows] = await getPool().execute<UserRow[]>(
    `SELECT id, email, display_name, photo_url, email_verified_at, status, is_admin, account_role,
            avatar, has_received_welcome_notification
       FROM users WHERE id = ? LIMIT 1`,
    [userId],
  );
  if (!rows.length || rows[0].status !== 'active') return null;

  const user = rows[0];
  return {
    uid: user.id,
    email: user.email,
    displayName: user.display_name || user.email.split('@')[0],
    photoURL: user.photo_url,

    avatar: user.avatar || 'avatar1',
    hasReceivedWelcomeNotification: Boolean(user.has_received_welcome_notification),
  };
}

export async function hashPassword(password: string): Promise<string> {
  const salt = randomBytes(16).toString('hex');
  const derivedKey = (await scrypt(password, salt, 64)) as Buffer;
  return `scrypt$${salt}$${derivedKey.toString('hex')}`;
}

export async function verifyPassword(password: string, storedHash: string): Promise<boolean> {
  const [algorithm, salt, expectedHex] = storedHash.split('$');
  if (algorithm !== 'scrypt' || !salt || !expectedHex) return false;

  const expected = Buffer.from(expectedHex, 'hex');
  const actual = (await scrypt(password, salt, expected.length)) as Buffer;
  return expected.length === actual.length && timingSafeEqual(expected, actual);
}

  return randomBytes(32).toString('base64url');
}

function normalizeWebsiteUrl(value: unknown): string | null {
  if (typeof value !== 'string' || value.trim().length > 512) return null;
  try {
    const parsed = new URL(value.trim());
    if (!['http:', 'https:'].includes(parsed.protocol)) return null;
    if (parsed.username || parsed.password) return null;
    if (!parsed.hostname.includes('.') || ['localhost', '127.0.0.1'].includes(parsed.hostname.toLowerCase())) return null;
    return parsed.toString().replace(/\/$/, '');
  } catch {
    return null;
  }
}

import type { PoolConnection, ResultSetHeader } from 'mysql2/promise';

/**
 * Idempotently create the commission for a captured Razorpay payment.
 *
 * New referral records explicitly store UTC timestamps in verify-otp/route.ts.
 * Commission eligibility is based on the immutable referral association plus a
 * captured payment, not by comparing legacy DATETIME values: older records may
 * have been written with the database server's local timezone. The unique key
 * on payment_record_id plus INSERT IGNORE protects retries and concurrent calls.
 */
export async function ensureAffiliateCommissionForPayment(
  connection: PoolConnection,
  paymentRecordId: number,
): Promise<number> {
  const [result] = await connection.execute<ResultSetHeader>(
    `INSERT IGNORE INTO affiliate_commissions
       (affiliate_user_id, referred_user_id, payment_record_id,
        amount_paise, commission_percent, commission_paise, status)
     SELECT
        ar.affiliate_user_id,
        ar.referred_user_id,
        p.id,
        p.amount_paise,
        ap.commission_percent,
        FLOOR(p.amount_paise * ap.commission_percent / 100),
        'pending'
     FROM payments p
     JOIN affiliate_referrals ar
       ON ar.referred_user_id = p.user_id
      AND ar.status = 'qualified'
     JOIN affiliate_profiles ap
       ON ap.user_id = ar.affiliate_user_id
      AND ap.status = 'active'
     JOIN users au
       ON au.id = ap.user_id
      AND au.account_role = 'affiliate'
      AND au.status = 'active'
     WHERE p.id = ?
       AND p.provider = 'razorpay'
       AND p.status = 'captured'
       AND p.paid_at IS NOT NULL
     LIMIT 1`,
    [paymentRecordId],
  );

  return result.affectedRows;
}
